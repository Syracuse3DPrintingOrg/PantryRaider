# Every line here is an exact `==` pin, and service/requirements.lock is the
# hash-locked resolution of this file that the image actually installs. Change a
# pin and the lock has to be regenerated in the same commit (README:
# Dependencies); tests/test_requirements_lock.py fails the build otherwise.

# Web framework and server (required)
# fastapi 0.132.x is the newest line that still caps starlette below 1.0, so the
# resolution lands on a starlette carrying the 0.49.1 security fixes without
# dragging in a major-version rewrite of the ASGI layer under the app.
fastapi==0.132.1
uvicorn[standard]==0.30.6

# Database ORM for SQLite inventory (required)
sqlalchemy==2.0.35

# Data validation and settings management (required)
pydantic==2.9.2
pydantic-settings==2.5.2

# Async HTTP client - used for Grocy, Mealie, and cloud provider APIs (required)
httpx==0.27.2

# Form and file upload parsing for photo analysis endpoints (required).
# 0.0.31 or newer: every photo, receipt, and import upload route in this app
# parses a multipart body, and the releases after CVE-2024-53981 closed six
# more parser holes, the last in 0.0.31 (unbounded part headers, oversized
# preamble/epilogue, arbitrary file write, Content-Disposition and querystring
# parameter smuggling, and a negative Content-Length that buffers the whole
# body in memory).
python-multipart==0.0.32

# HTML templating engine for the web UI (required)
jinja2==3.1.6

# AI vision providers (optional - install whichever providers you use)
google-generativeai==0.8.3  # Gemini vision/enrichment provider
# Transitive dep of google-generativeai. Pin to a current normalized release so
# pip never resolves the legacy 1.0beta5prerelease2 wheel, whose non-normalized
# version pip 25.3 hard-errors on (pypa/pip#12938).
google-api-python-client==2.197.0
anthropic==1.4.0             # Anthropic Claude vision/enrichment provider
# OpenAI is accessed via httpx directly - no SDK needed

# Image resizing before sending to AI providers (required for photo analysis),
# and the label and document renderers. 12.3.0 or newer: the 10.4.0 line is
# open to a long list of malformed-image advisories, including heap
# out-of-bounds writes reachable from Image.paste / Image.crop, ImageCms, and
# the RankFilter this app's renderers use on uploaded photos.
Pillow==12.3.0

# PDF text extraction for the "Import from PDF" recipe upload. Pure Python (no
# poppler or other system deps); the recipe's text is handed to the same LLM
# extractor the URL import fallback uses.
pypdf==6.14.2

# Renders a scanned / image-only PDF's pages to images so the vision AI can read
# them (the "Import from PDF" fallback). Its wheels bundle the PDFium engine and
# ship for both amd64 and arm64 (manylinux2014 x86_64 + aarch64, plus armv7l),
# so there is still no poppler or other system binary to install in the image.
pypdfium2==5.13.0

# Structured-data recipe scraping for the "From URL" import (the same library
# Mealie wraps, now run in-process for the native recipe store). Pure Python;
# pages the scraper cannot read fall back to the existing LLM extraction.
recipe-scrapers==15.11.0

# Flexible date parsing for expiry date extraction (required)
python-dateutil==2.9.0

# Signed session cookies for browser authentication (required)
itsdangerous==2.2.0

# TOTP two-factor authentication (required)
pyotp==2.10.0

# QR code generation for TOTP setup page and phone deep-link (required).
# [pil] pulls Pillow for the PNG TOTP code; SVG output uses qrcode's built-in
# stdlib SVG factory in 8.x, so no extra (and no lxml) is needed.
qrcode[pil]==8.2

# mDNS advertisement so the Home Assistant integration can find this install
# on the LAN without the owner typing an IP into the config flow (required).
# Pure Python; pulls in ifaddr transitively for interface enumeration.
zeroconf==0.150.0
