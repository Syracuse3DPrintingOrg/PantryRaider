# Web framework and server (required). Starlette is pinned explicitly (it
# arrives transitively) and has to move together with fastapi: 1.3.1 or later
# caps a urlencoded form at 1000 fields, which bounds FastAPI's per-key form
# parsing, and it carries the URL and form security fixes. Newer fastapi on an
# older Starlette lets a few hundred kilobytes of junk form keys posted to
# /signup hold the worker for tens of seconds.
fastapi==0.141.1
starlette==1.7.0
uvicorn[standard]==0.54.0

# Database ORM (required); psycopg2 is the production Postgres driver.
# Tests run on SQLite and do not need it.
sqlalchemy==2.1.1
psycopg2-binary==2.9.13

# Schema migrations. Startup runs `alembic upgrade head` for a database
# already under Alembic control; see migrations/README.md.
alembic==1.20.0

# Settings management (required)
pydantic==2.13.5
pydantic-settings==2.15.0

# Templates for the web portal (required). 3.1.6 carries the 2024-2025
# security fixes (sandbox escapes and the xmlattr injection advisory).
jinja2==3.1.6

# Async HTTP client, for the AI forwarder once a provider is wired up
httpx==0.28.1

# Form and file upload parsing for the AI proxy endpoint and the backup
# upload (required). Forager takes multipart from the open internet, so this
# floor matters more here than it does on a LAN appliance: 0.0.31 closes a run
# of parser advisories after 0.0.18, including unbounded part headers, an
# oversized preamble, an arbitrary file write, and Content-Disposition and
# querystring smuggling.
python-multipart==0.0.32

# QR codes for two-factor sign-in enrollment. The SVG image factory renders
# the authenticator QR with no Pillow dependency, so it stays light.
qrcode==8.2

# Passkey (WebAuthn / FIDO2) sign-in for the Forager account. The maintained
# py_webauthn package builds the registration/authentication options and
# verifies attestations and assertions, so the app never hand-rolls CBOR/COSE.
# It pulls in cryptography and cbor2; both are pure wheels on the target arch.
webauthn==2.8.0

# Fernet (symmetric authenticated encryption) for the two-factor authenticator
# seed at rest (app/totp_crypto.py). Already present transitively via webauthn;
# pinned explicitly so this security dependency is intentional, not incidental.
cryptography==50.0.1

# Reading text out of an uploaded recipe PDF on the portal upload path. Pure
# Python, no native deps; a scanned (image-only) PDF yields no text and the
# upload path falls back to a "take a photo instead" message. The 6.x line is
# required: it carries the FlateDecode and LZWDecode fixes for the crafted
# PDFs that could exhaust memory in 5.x, and this parses stranger-supplied
# files. The API used here (PdfReader, .pages, .extract_text) is unchanged.
pypdf==6.19.0
